// buildMergedPreset 补全合并
function buildMergedPreset(preset, job) {
  // ... existing border merge ...
  const mergedInfo = Object.assign({}, preset.infoBar, job.infoOverrides);
  // 确保所有字段被合并
  ['titleFontSize','subtitleFontSize','fontColor','textAlign','extraLines',
   'titleFont','subtitleFont','titleGap','textOffsetX','textOffsetY'].forEach(k => {
    if (job.infoOverrides[k] !== undefined) mergedInfo[k] = job.infoOverrides[k];
  });
  // ...
}
